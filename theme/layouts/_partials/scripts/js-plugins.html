{{/* JS-plugin loop (js-plugins spike): emits each registered plugin from assets/js/plugins/ (theme, project, or module-mounted; a same-named project file shadows the theme's). Registry: params.docsy.jsPlugins, a list of { name, enable, defer, pageGate, options }. Options are passed to js.Build as params and reach the module via `import * as params from '@params'`. Registry order is the emission order. pageGate names a .Page.Store flag; when set, the plugin is emitted only on pages carrying the flag. An entry may name a vendorPartial (under scripts/), emitted right after its plugin script — the vendored-library hook. */ -}} {{/* Theme-default entries, aliasing legacy params: params.markmap.* keeps working for one deprecation cycle by mapping onto a registry entry; a site-declared markmap entry overrides the alias. */ -}} {{ $registry := .Site.Params.docsy.jsPlugins | default slice -}} {{ $names := slice -}} {{ range $registry }}{{ $names = $names | append .name }}{{ end -}} {{ if and .Site.Params.markmap.enable (not (in $names "markmap")) -}} {{ $options := dict "height" (.Site.Params.markmap.height | default "300px") -}} {{ $registry = $registry | append (dict "name" "markmap" "options" $options "vendorPartial" "markmap-vendor.html") -}} {{ end -}} {{ if not (in $names "tabpane-persist") -}} {{ $registry = $registry | append (dict "name" "tabpane-persist" "pageGate" "hasTabs") -}} {{ end -}} {{ if and (not .Site.Params.prism_syntax_highlighting) (not .Site.Params.disable_click2copy_chroma) (not (in $names "click-to-copy")) -}} {{ $registry = $registry | append (dict "name" "click-to-copy" "defer" true) -}} {{ end -}} {{ range $registry -}} {{ $entry := . -}} {{ if ne $entry.enable false -}} {{ if or (not $entry.pageGate) ($.Page.Store.Get $entry.pageGate) -}} {{ $path := printf "js/plugins/%s.js" $entry.name -}} {{ with resources.Get $path -}} {{ $opts := dict "params" ($entry.options | default dict) -}} {{ if hugo.IsProduction -}} {{ $opts = merge $opts (dict "minify" true) -}} {{ end -}} {{ $js := . | js.Build $opts -}} {{ if hugo.IsProduction -}} {{ $js = $js | fingerprint -}} {{ else -}} {{ end -}} {{ with $entry.vendorPartial -}} {{ partial (printf "scripts/%s" .) $ -}} {{ end -}} {{ else -}} {{ warnidf "docsy-js-plugin-missing" "JS plugin %q is registered in params.docsy.jsPlugins but assets/%s was not found." $entry.name $path -}} {{ end -}} {{ end -}} {{ end -}} {{ end -}}
